q_symgamma <- function(u, alpha, gamma) {
  pos_part <- qgamma(2 * pmax(u - 0.5, 0), shape = alpha, rate = gamma)
  neg_part <- -qgamma(2 * pmax(0.5 - u, 0), shape = alpha, rate = gamma)
  ifelse(u > 0.5, pos_part, neg_part)
}

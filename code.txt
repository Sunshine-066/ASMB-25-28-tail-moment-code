#Method 2: Maximum Likelihood Estimation
fgm_mle_copula <- function(data) {
  # Transform data to uniform scores (using empirical distribution function)
  u <- rank(data$sedan) / (length(data$sedan) + 1)
  v <- rank(data$stnwg) / (length(data$stnwg) + 1)
  
  # Density function of FGM copula
  # c(u,v) = 1 + θ(1-2u)(1-2v)
  fgm_copula_density <- function(theta, u, v) {
    return(1 + theta * (1 - 2*u) * (1 - 2*v))
  }
  
  # Negative log-likelihood function
  neg_log_likelihood <- function(theta) {
    # Ensure theta is within [-1,1]
    if(abs(theta) > 1) return(1e10)
    
    # Calculate density values
    c_vals <- fgm_copula_density(theta, u, v)
    
    # Ensure density is positive
    if(any(c_vals <= 0)) return(1e10)
    
    # Return negative log-likelihood
    return(-sum(log(c_vals)))
  }
  
  # Use optimize for one-dimensional optimization
  result <- optimize(neg_log_likelihood, interval = c(-1, 1))
  
  # Calculate standard error (using numerical Hessian matrix)
  hessian_val <- hessian(neg_log_likelihood, result$minimum)
  se <- sqrt(1 / hessian_val)
  
  return(list(
    theta = result$minimum,
    log_likelihood = -result$objective,
    standard_error = se,
    converged = TRUE
  ))
}

# Gamma distribution moment estimation function
gamma_mom <- function(data) {
  m <- mean(data)
  v <- var(data)
  shape <- m^2 / v         # α (shape parameter)
  rate <- m / v            # γ (inverse of scale parameter)
  return(list(shape = shape, rate = rate, 
              mean = m, var = v))
}
